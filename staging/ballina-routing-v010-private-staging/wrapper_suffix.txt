const __ballinaCache = {};
function __ballinaRequire(moduleId) {
  const cached = __ballinaCache[moduleId];
  if (cached) return cached.exports;
  const factory = __ballinaModules[moduleId];
  if (!factory) throw new Error('unknown_module');
  const module = {exports: {}};
  __ballinaCache[moduleId] = module;
  factory(module, module.exports, __ballinaRequire);
  return module.exports;
}

const constantTimeToken = async (provided) => {
  const expected = Deno.env.get('BALLINA_V010_STAGING_TOKEN') ?? '';
  if (!expected || expected.length < 32 || expected === privateKey || expected === publishableKey || !provided || provided.length > 256) return false;
  const digest = async s => new Uint8Array(await crypto.subtle.digest('SHA-256',new TextEncoder().encode(s)));
  const [a,b] = await Promise.all([digest(expected),digest(provided)]);
  let diff=0; for(let i=0;i<a.length;i++) diff|=a[i]^b[i];
  return diff===0;
};

  return '';
}

function permittedByLocalLimit() {
  const tick=Date.now(), slot=Math.floor(tick/60000);
  for (const k of limits.keys()) if (k!==slot) limits.delete(k);
